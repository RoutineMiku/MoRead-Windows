  async function copyText(text) {
    try {
      if (navigator.clipboard && navigator.clipboard.writeText) {

        return true;
      }
    } catch (e) { /* 继续走兜底 */ }
    try {
      const ta = document.createElement('textarea');
      ta.value = text;
      ta.setAttribute('readonly', 'readonly');
      ta.style.cssText = 'position:fixed;left:-9999px;top:0;opacity:0';
      document.body.appendChild(ta);
      ta.select();
      ta.setSelectionRange(0, ta.value.length);
      const ok = document.execCommand('copy');
      document.body.removeChild(ta);
      return !!ok;
    } catch (e) { return false; }
  }

  /* 书架只做展示排序，不改动存档里 books 的顺序（导出的书架文件顺序保持原样） */
  function sortBooks(list) {
    const pct = b => pctOf(b);
    /* 旧存档只有 progress.at，没有 book.readAt：两种都认，「最近阅读」才不会把老书全排到后面 */
    const lastRead = b => b.readAt || (progressOf(b.id) && progressOf(b.id).at) || 0;
    const byRecent = (a, b) => lastRead(b) - lastRead(a) || (b.createdAt || 0) - (a.createdAt || 0);
    const key = shelfSortKey();
    const cmp = {
      recent: byRecent,
      added: (a, b) => (b.createdAt || 0) - (a.createdAt || 0) || zhCompare(a.name || '', b.name || ''),
      name: (a, b) => zhCompare(a.name || '', b.name || ''),
      author: (a, b) => zhCompare(a.author || '佚名', b.author || '佚名') || zhCompare(a.name || '', b.name || ''),
      progress: (a, b) => pct(b) - pct(a) || byRecent(a, b),
      updated: (a, b) => (b.updatedAt || 0) - (a.updatedAt || 0) || byRecent(a, b)
    }[key] || byRecent;
    return list.slice().sort(cmp);
  }

function decodeBuffer(buf) {
  if (buf.length > 3 && buf[0] === 0xEF && buf[1] === 0xBB && buf[2] === 0xBF) {
    return new TextDecoder('utf-8').decode(buf.subarray(3));
  }
  if (buf.length > 1 && buf[0] === 0xFF && buf[1] === 0xFE) {
    return new TextDecoder('utf-16le').decode(buf.subarray(2));
  }
  try {
    return new TextDecoder('utf-8', { fatal: true }).decode(buf);
  } catch (e) {
    return new TextDecoder('gb18030').decode(buf);
  }
}

const ENT = { nbsp: ' ', amp: '&', lt: '<', gt: '>', quot: '"', apos: "'", ldquo: '“', rdquo: '”', mdash: '—', hellip: '…', middot: '·' };
function decodeEntities(s) {

function bytesToBase64(bytes) {
  if (typeof Buffer !== 'undefined' && Buffer.from) return Buffer.from(bytes).toString('base64');
  let s = '';
  for (let i = 0; i < bytes.length; i += 0x8000) {

  }

}
function utf8Bytes(str) {
  if (typeof Buffer !== 'undefined' && Buffer.from) return Buffer.from(str, 'utf8');
  if (typeof TextEncoder !== 'undefined') return new TextEncoder().encode(str);

}

function parseLocalFile(filePath) {
  const fs = require('fs');
  const buf = fs.readFileSync(filePath);
  const ext = path.extname(filePath).toLowerCase();
  if (ext === '.epub') {
    const r = parseEpub(buf);
    return { kind: 'epub', title: r.title, author: r.author, chapters: r.chapters };
  }
  const text = decodeBuffer(buf);
  const chapters = splitTxt(text);
  if (!chapters.length) throw new Error('文件里没有可读文本');
  return { kind: 'txt', title: path.basename(filePath, ext), author: '', chapters };
}

function cacheSize() {

    let list = [];

    for (const f of list) {
      const p = path.join(d, f.name);
      if (f.isDirectory()) walk(p);

    private fun cssPx(raw: Int): Int =
        Math.round(raw / resources.displayMetrics.density.toFloat()).coerceIn(0, 64)
    private fun collectInsets() {
        val root: View = window.decorView
        ViewCompat.setOnApplyWindowInsetsListener(root) { v, ins ->
            val bars = WindowInsetsCompat.Type.systemBars() or WindowInsetsCompat.Type.displayCutout()
            val t = ins.getInsets(bars)
            val top = cssPx(t.top)
            val bottom = cssPx(t.bottom)
            val changed = lastInsets.optInt("top") != top || lastInsets.optInt("bottom") != bottom
            lastInsets = JSONObject()
                .put("top", top).put("bottom", bottom)
                .put("left", cssPx(t.left)).put("right", cssPx(t.right))
            /* 立刻推给页面。启动时 syncLayout 读到的 lastInsets 还是空的，
             * 回调往往晚一两帧才来 —— 不补这一枪，--sa-top 会永远是 0。 */
            if (changed) {
                bridge.emitToJs("if(window.MoApp&&MoApp.onInsets){MoApp.onInsets(${lastInsets})}")
            }
            pushLayout()
            ViewCompat.onApplyWindowInsets(v, ins)
            ins
        }
    }

    fun pickFiles(mimes: List<String>, multiple: Boolean): List<Picked> {

            pickSink = { uris ->

            }
            runCatching {
                pickLauncher.launch(Pair(if (mimes.isEmpty()) arrayOf("*/*") else mimes.toTypedArray(), multiple))
            }.onFailure { latch.countDown() }
        }
        latch.await(3, TimeUnit.MINUTES)
        return result.map { uri ->
            takePermission(uri)
            runCatching {
                Picked(uri, queryName(uri), querySize(uri), MoFiles.stage(this, uri.toString()), null)
            }.getOrElse { e ->
                // 不能静默丢弃：用户选了文件却什么都没发生，是最难自查的一类问题
                Log.w(TAG, "暂存文件失败: ${e.message}")
                Picked(uri, queryName(uri), querySize(uri), "", e.message ?: "读取失败")
            }
        }
    }
